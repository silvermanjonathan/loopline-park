# ==========================================================
#  LOOPLINE PARK - COASTER SIMULATOR  (turtle edition)
#  Same track, same physics, nothing to install. The train
#  steps from tower to tower instead of gliding.
# ==========================================================

import turtle
import math
import time

# --- YOUR TRACK -------------------------------------------
# Each number is a hill height in meters, left to right.
# Rule 1: the FIRST number is the lift hill. It must be the tallest.
# Rule 2: change these numbers and rerun. That IS the design job.

track = [50, 48, 42, 33, 22, 11, 2, 0, 8, 18, 26, 30, 26, 18, 9, 3, 0]

SEG_LEN = 20          # each hill tower sits 20 meters from the next
GRAVITY = 9.8

# --- SETUP ------------------------------------------------
screen = turtle.Screen()
screen.setup(900, 600)
screen.bgcolor("#1B2951")
screen.title("Loopline Park - Coaster Simulator (turtle)")

step_x = 800 / (len(track) - 1)
scale_y = 400 / max(track)

pen = turtle.Turtle()
pen.hideturtle()
pen.speed(0)
pen.pensize(4)
pen.color("#FCD34D")
pen.penup()
pen.goto(-400, track[0] * scale_y - 250)
pen.pendown()
for i in range(len(track)):
    pen.goto(-400 + i * step_x, track[i] * scale_y - 250)

train = turtle.Turtle()
train.shape("square")
train.color("#DC2626")
train.penup()
train.goto(-400, track[0] * scale_y - 240)

dash = turtle.Turtle()
dash.hideturtle()
dash.penup()
dash.color("#F8FAFC")
dash.goto(-390, 240)

# --- RUN THE RIDE -----------------------------------------
top = track[0]
speed = 2.0
stalled = 0

for i in range(len(track)):
    height = track[i]            # at a tower, the height IS the list number

    # THE PHYSICS: height you have lost becomes speed you have gained
    drop = top - height
    if drop < 0:
        stalled = 1
    else:
        speed = math.sqrt(2 * GRAVITY * drop)
        if speed < 2:
            speed = 2            # the chain lift lets go at 2 m/s

    if stalled == 1:
        dash.clear()
        dash.write("STALLED - hill taller than the lift!", font=("Courier", 18, "bold"))
        break                    # engine word: ends the loop early

    train.goto(-400 + i * step_x, track[i] * scale_y - 240)
    dash.clear()
    dash.write(f"SPEED {speed * 2.24:.0f} mph    HEIGHT {height:.0f} m", font=("Courier", 16, "bold"))
    time.sleep(SEG_LEN / speed / 4)

turtle.done()

# ==========================================================
#  LOOPLINE PARK - COASTER SIMULATOR - STAGE 3
#  First motion. The physics, with nothing else. Will it move?
# ==========================================================

import pygame
import math

track = [50, 48, 42, 33, 22, 11, 2, 0, 8, 18, 26, 30, 26, 18, 9, 3, 0]

SEG_LEN = 20          # each hill tower sits 20 meters from the next
GRAVITY = 9.8
FPS = 60
DT = 1.0 / FPS        # how much time passes in one frame

NAVY = (27, 41, 81)
YELLOW = (252, 211, 77)
RED = (220, 38, 38)
CREAM = (248, 250, 252)
GREEN = (16, 185, 129)

pygame.init()
screen = pygame.display.set_mode((900, 600))
pygame.display.set_caption("Loopline Park - Coaster Simulator")
clock = pygame.time.Clock()
big = pygame.font.SysFont("couriernew", 30, bold=True)
small = pygame.font.SysFont("couriernew", 18)

tallest = max(track)
step_x = 800 / (len(track) - 1)
scale_y = 400 / tallest

rail_points = []
for i in range(len(track)):
    x = 50 + i * step_x
    y = 550 - track[i] * scale_y
    rail_points.append((x, y))

# --- RUN THE RIDE -----------------------------------------
top = track[0]
position = 0.0                       # how many meters along the track we are
track_length = (len(track) - 1) * SEG_LEN
speed = 2.0

for frame in range(3000):
    pygame.event.pump()              # keeps the window alive

    seg = int(position / SEG_LEN)
    if seg > len(track) - 2:
        seg = len(track) - 2
    frac = (position - seg * SEG_LEN) / SEG_LEN

    height = track[seg] + (track[seg + 1] - track[seg]) * frac

    # THE PHYSICS: height you have lost becomes speed you have gained
    drop = top - height
    speed = math.sqrt(2 * GRAVITY * drop)

    # move forward by however fast we are going
    position = position + speed * DT

    screen.fill(NAVY)
    pygame.draw.line(screen, GREEN, (50, 550), (850, 550), 3)
    pygame.draw.lines(screen, YELLOW, False, rail_points, 4)

    for i in range(len(track)):
        pygame.draw.line(screen, (60, 80, 130),
                         (rail_points[i][0], rail_points[i][1]),
                         (rail_points[i][0], 550), 2)

    tx = 50 + (position / SEG_LEN) * step_x
    ty = 550 - height * scale_y
    pygame.draw.rect(screen, RED, (tx - 16, ty - 14, 32, 14), border_radius=4)

    screen.blit(big.render(f"SPEED  {speed * 2.24:.0f} mph", True, CREAM), (50, 30))
    screen.blit(small.render(f"HEIGHT {height:.0f} m", True, CREAM), (50, 70))
    screen.blit(small.render(f"TIME   {frame * DT:.1f} s", True, CREAM), (50, 92))

    pygame.display.flip()
    clock.tick(FPS)

    if position >= track_length:
        break

pygame.time.wait(3000)
pygame.quit()

The one thing square roots refuse to do

Your Stage 4 engine has a hidden weakness. math.sqrt refuses negative numbers — run print(math.sqrt(-1)) in a scratch file and it crashes with ValueError: math domain error. On the stock track, drop never goes below zero… but you are about to design your own tracks. Stage 5 posts an inspector at the door:

    drop = top - height
    if drop < 0:
        stalled = 1
    else:
        speed = math.sqrt(2 * GRAVITY * drop)
        if speed < 2:
            speed = 2

    if stalled == 0:
        position = position + speed * DT
Predict Read the inspector's lines. stalled is the engine's light switch — 0 running, 1 frozen — and if stalled == 0 (that's == asking, Tuesday's question mark) is the only gate that lets the train move. ↓

Chunk 8 · The debugging protocolE5-PRO-TR-?? ⚠

Read the error first

During the build, things will break. That's the plan working. This crash comes from a real Loopline machine — the very engine you just ran. Line 1 of the file says import math. Deep inside, one line reads:

        speed = sqrt(2 * GRAVITY * drop)
Traceback (most recent call last):
  File "coaster.py", line 71, in <module>
    speed = sqrt(2 * GRAVITY * drop)
NameError: name 'sqrt' is not defined
Predict Which line of the traceback do you read first? import math is right there at the top of the file — so what does Python not recognize, and what's the fix?
Last line first — the week's oldest rule. NameError: the same species as Monday's pint. Importing math carries the toolbox into the room, but you still have to name the drawer — the function's full name is math.sqrt, and on its own sqrt names nothing. The computer does not guess: day one's rule, still the whole game on day three. Fix: speed = math.sqrt(2 * GRAVITY * drop). (Today's likeliest crash in your own booth is humbler — a missing : on an if line, SyntaxError — same protocol, last line first.) During the build, keep your two-line debugging log for every crash: what broke, what fixed it. Nobody fixes your code for you today — the log is proof it was you.

The Grand Opening build

Build what your approved blueprint describes. The contract has two halves:

The booth (your program): one screen, one loop, one decision, constructs from at least three prior days. It runs, and it matches the blueprint.

The ride (your data): a track list of at least 10 towers — first tower is your lift hill and the tallest, last tower is 0, and it completes with no STALLED banner (you know why from Chunk 3). Save it, then change at least three numbers with a goal — faster? gentler? longer? — and save again. Version 1 and Version 2 both kept, with the one-sentence what-changed-and-why note on your Ride Log.

Your toolbox from the week: print() / input() / int() / f-strings (Day 1) · for + range(), turtle (Day 1) · if / elif / else, lists (Day 2) · random.choice(), def (Day 2).

Suggested start: build your blueprint's first box only. Run it. Then the next box. Run after every box — small steps leave small bugs. The engine pastes in below your booth, and the Coaster Lab page holds the full Track Design Card if you want tested shapes to start from.

# THE OPERATOR'S BOOTH - Loopline Park Grand Opening
import pygame
import math

thrill_track = [50, 46, 30, 8, 0, 14, 30, 44, 20, 0, 12, 0]
family_track = [50, 47, 44, 40, 36, 33, 36, 40, 36, 33, 31, 30]

rides = ["The Python Plunge", "The Family Cruiser"]

name = input("Welcome to the Grand Opening! Your name? ")
print(f"Hello, {name}! Today's rides:")
for ride in rides:
    print("- " + ride)

mood = input("Feeling thrill or chill today? ")
if mood == "thrill":
    track = thrill_track
    print(f"Boarding pass: {name} - THE PYTHON PLUNGE. Hold on!")
else:
    track = family_track
    print(f"Boarding pass: {name} - THE FAMILY CRUISER. Enjoy the view!")

Day 1 (input, f-strings) + Day 2 (loop) + Day 3 (two lists, decision) — three days before the engine even starts. Both tracks are tested: the Python Plunge completes in 13.0 seconds at 70 mph; the Family Cruiser never drops below 30 m and cruises 17.8 seconds at 44 mph — proof that gentle is a design choice you make with numbers. Two things to notice: the else is a safety net — a guest who types anything unexpected gets the family ride instead of a crash — and one engine line must go: the engine's own track = [...] line, because your booth already decided. For extra flavor, add a third option where track gets chosen by random.choice() from a list of lists — Tuesday's spinner, spinning whole coasters: the mystery ride.

3:30 · Soft Launch — the safety inspection

One guest, before the public

Real parks never open a ride to the public first — they soft-launch it for inspectors. Today's inspector is one partner:

1. Run your booth live, with your partner as the guest — their answers, not yours.
2. Invite them to try to break it: type something unexpected at the mood question, ride the ride, read the dashboard.
3. As your ride runs, rehearse the narration: the three physics lines, on your own track.
4. Log everything that broke, wobbled, or confused your inspector on your Maintenance slip. Nothing has to be perfect today — it has to be diagnosed today. Tomorrow is Expansion Day — the fix-it clinic and the upgrade garage — and Friday at 3:25 the ribbon gets cut for the whole room.

Choose your exit gate

Floor · approaching the standard

The stock ride runs on your machine (pygame or turtle), your Ride Log Grid A matches the dashboard at every marked tower, your blueprint is approved, and your booth runs with a sequence plus a loop or a decision.

Middle · meets the standard

Your booth follows your approved blueprint — sequence, a loop, and a decision that chooses the track, using constructs from at least three days. Your own ride completes, Version 1 and Version 2 saved with the what-changed note — and at today's soft launch you narrate the three physics lines — Friday you do it for the room.

Stretch · exceeds the standard

Your booth survives a guest who types something unexpected without crashing (the sample's else safety net) — and your ride carries one engineering upgrade you can explain: the friction line from the Coaster Lab's stretch gate (drop = top - height - position * 0.01, tuned), a mystery track drawn by random.choice(), or a second decision (a fast-pass lane). Friday at the walkthrough, name the design you rejected, and why.
